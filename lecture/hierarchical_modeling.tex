\renewcommand{\frametitleText}{Simple example: random means/intercepts model}
\begin{frame}
\frametitle{\frametitleText}
Recall the model from lecture on regularization from Bayes lens:
\begin{tightEquation*}
y_i \given \mu_i
	\sim \normalDist(\mu_i, \phi^{-1}\! = \sigma^2), \ \,
\mu_i \given \phi_{\textrm{grp}} 
	\sim \normalDist( 
		\mu_\mathrm{grp}, \thinnerspace 
		\phi_{\textrm{grp}}^{-1}\! = \sigma_{\textrm{grp}}^2
	).
\end{tightEquation*}

\smallskip
Here $\mu_i$'s dist is controlled by ``higher-level'' param $(\mu_\mathrm{grp}, \phi_{\mathrm{grp}})$:
\vspace*{-.4\baselineskip}
\begin{figure} 
\begin{tikzpicture}
  % Define nodes
  \node[obs] (y) {$y_i\!$};
  \node[latent, right=1cm of y] (mu) {$\mu_i\!$};
  \node[const, right=1cm of mu] (phi) {$\, \phi_{\mathrm{grp}} \, $};
  \coordinate[right=.3cm of mu] (muRight);

  % Connect the nodes
  \edge {mu} {y} ; %
  \edge {phi} {mu} ; %

  % Plates
  \plate [inner sep=.3cm, yshift=.15cm] {caption} {(y)(mu) (muRight)} {$i = 1, \ldots, n\!$} ;
\end{tikzpicture}
\end{figure}
\end{frame}


\begin{frame}
\frametitle{\frametitleText}
We saw that this hierarchical structure yields shrinkage estimator:
\begin{tightEquation*}
\mu_i \given y_i, \phi_\mathrm{grp} 
	\sim \normalDist\!\left(
		\frac{1}{\phi + \phi_\mathrm{grp}} 
			( \phi \thinnerspace y_i + \phi_\mathrm{grp} \thinnerspace \mu_\mathrm{grp} ), \thinnerspace
		\frac{1}{\phi + \phi_\mathrm{grp}}
	\right),
\end{tightEquation*}
\pause%
where plugging in the maximizer $\hat{\phi}_\mathrm{grp}$ of the marginal likelihood
\begin{tightEquation*}
\likelihood(\by \given \phi_{\mathrm{grp}})
	= \int \likelihood(\by \given \bmu, \phi_{\mathrm{grp}}) \thinnerspace \density(\bmu \given \phi_{\mathrm{grp}}) \diff  \bmu
	\sim \normalDist(\bm{0}, \phi_{\mathrm{mg}}^{-1} \Id )
\end{tightEquation*}
for $\phi_{\mathrm{mg}}^{-1} = \phi^{-1} + \phi_{\mathrm{grp}}^{-1}$ yielded a near-frequentist-optimal answer.

\pause%
\vspace*{9pt}
Fully Bayesian inference is obtained by placing a prior on $\phi_\mathrm{grp}$ and integrating over $\density( \phi_\mathrm{grp} \given \by )$ (for which you can now utilize \mcmc{}).
\end{frame}


\begin{frame}
\frametitle{\frametitleText}
Let's consider a slightly more general random intercept model.

\pause%
We start by reinterpreting the previous one as a group-mean model where each of the groups $\thinnerspace j = 1, \ldots, J \thinnestspace$ has one member:
\begin{tightEquation*}
y_j \given \mu_j
	\sim \normalDist(\mu_j, \sigma^2), \ \,
\mu_j \given \sigma_{\textrm{grp}} 
	\sim \normalDist( 
		\mu_\mathrm{grp}, \thinnerspace 
		\sigma_{\textrm{grp}}^2
	),
\end{tightEquation*}
now parametrized in terms of the variance $\sigma^2$ and $\sigma_{\textrm{grp}}^2$.
\end{frame}


\begin{frame}
\frametitle{\frametitleText}
Let's now imagine the $j$th group having $i = 1, \ldots, n_j$ members:
\begin{tightEquation*}
y_{\highlight{i}j} \given \mu_j
	\sim \normalDist(\mu_j, \sigma^2), \ \,
\mu_j \given \sigma_{\textrm{grp}} 
	\sim \normalDist( 
		\mu_\mathrm{grp}, \thinnerspace 
		\sigma_{\textrm{grp}}^2
	),
\end{tightEquation*}
\pause%
or equivalently in a regression model language,
\begin{tightEquation*}
y_{ij} 
	= \mu_\mathrm{grp} + \randEff_{j} + \epsilon_{ij}, \ \,
\randEff_{j} 
	\sim \normalDist( 
		0, \sigma_{\textrm{grp}}^2
	), \ \,
\epsilon_{ij}
	\sim \normalDist( 
		0, \sigma^2 
	).
\end{tightEquation*}

\pause%
\smallskip
The group-mean model can also be written in a ``tall'' format as
\begin{tightEquation*}
y_{i} \given \bm{\mu} % \mu_1, \ldots, \mu_J
	\sim \normalDist(\mu_{j(i)}, \sigma^2)
	\ \text{ for } \
	\textstyle i = 1, \ldots, \sum_{j = 1}^J n_j,
\end{tightEquation*}
where $j(i)$ indicates the membership of the $i$th observation $y_i$.\\
\pause%
Or as a regression model, with a $\bigl( \sum_{j = 1}^J n_j \bigr)$-by-$J$ binary matrix $\indicator$,
\begin{tightEquation*}
\by 
	= \bm{1} \mu_\mathrm{grp} 
	+ \indicator \bm{\randEff}
	+ \bm{\epsilon}, \ \,
	\bm{\randEff} 
		\sim \normalDist( 
			0, \sigma_{\textrm{grp}}^2 \bm{I}
		), \ \,
	\bm{\epsilon}
		\sim \normalDist( 
			0, \sigma^2 \bm{I}
		).
\end{tightEquation*}
\end{frame}

\begin{frame}
\frametitle{\frametitleText}
\begin{minipage}[t][.23\paperheight]{\linewidth}
\vspace*{-2.4\baselineskip}
{%
	\setlength{\belowdisplayskip}{.55\baselineskip}%

\begin{minipage}[t][.4\paperheight]{\linewidth}
\vspace*{-.125\baselineskip}
\fontdimen2\font=2.75pt
Can be written in the language of mixed-effect modeling as
\begin{tightEquation*}
y_z^\mathrm{zp} 
	= \mu^\mathrm{us}
	+ \randEff^\mathrm{st}_{s(c(z))} 
	+ \randEff^\mathrm{co}_{c(z)}
	+ \hat{\sigma}^\mathrm{zp}_z \epsilon_z
\end{tightEquation*}
where
$\,\randEff^\mathrm{st}_s \sim \normalDist(0, \sigma^\mathrm{st})$,
$\randEff^\mathrm{co}_c \sim \normalDist(0, \sigma^\mathrm{co})$,
and
$\, \epsilon_z \sim \normalDist(0, 1)$.
\end{minipage}
\end{frame}


\begin{frame}
\frametitle{\frametitleText}
Note that the mixed-effect model can also be written as
\begin{tightEquation*}
\setlength{\abovedisplayskip}{.6\baselineskip}
\begin{aligned}
y_z^\mathrm{zp} 
	&= \mu^\mathrm{us}
	+ \sum_{s'} \indicator_{ \{ s(c(z)) = s' \} } \randEff^\mathrm{st}_{s'} 
	+ \sum_{c'} \indicator_{ \{ c(z) = c' \} } \randEff^\mathrm{co}_{c'} 
	+ \hat{\sigma}^\mathrm{zp}_z \epsilon_z \\
	\stepcounter{beamerpauses}%
	\uncover<\value{beamerpauses}->{%
	&= \mu^\mathrm{us} 
	+ \bindicator^\mathrm{st}_z \bm{\randEff}^\mathrm{st}
	+ \bindicator^\mathrm{co}_z \bm{\randEff}^\mathrm{co}
	+ \hat{\sigma}^\mathrm{zp}_z \epsilon_z.
	}%
\end{aligned}
\end{tightEquation*}
%\addtocounter{beamerpauses}{-1}%

\pause
So the model in matrix form is given as:
\begin{tightEquation*}
\by^\mathrm{zp} 
	= \bm{1} \thinnerspace \mu^\mathrm{us}
	+ \bindicator^\mathrm{st} \bm{\randEff}^\mathrm{st}
	+ \bindicator^\mathrm{co} \bm{\randEff}^\mathrm{co}
	+ \bm{\epsilon}^\mathrm{zp}
\end{tightEquation*}
where 
$\thinnerspace \bm{\randEff}^\mathrm{st} \negthinnerspace 
	\sim \normalDist(\bm{0}, \sigma^\mathrm{st} \Id)$,
$\bm{\randEff}^\mathrm{co} \negthinnerspace 
	\sim \normalDist(\bm{0}, \sigma^\mathrm{co} \Id)$,
and
$\thinnerspace \bm{\epsilon}^\mathrm{zp} \negthinnerspace 
	\sim \normalDist(\bm{0}, \hat{\bSigma}^\mathrm{zp})$;
$\thinnerspace \bindicator^\mathrm{st} \!\in \mathbb{R}^{41,552 \times 50}$ and $\bindicator^\mathrm{co} \!\in \mathbb{R}^{41,552  \times 3,244} \thinnerspace$ are binary design matrices.

\pause
\smallskip
The hierarchical mean model is thus a special case of:
\begin{tightEquation*}
``\by
	= \bX \bbeta
	+ \bZ \bm{\randEff} 
	+ \bm{\epsilon}."
\end{tightEquation*}
\end{frame}


\section*{%
	\centerline{Linear mixed effect model:}
	\centerline{frequentist and Bayes perspectives}%
}%

\renewcommand{\frametitleText}{Classical perspective on linear mixed effect model}
\begin{frame}
\frametitle{\frametitleText}
Linear mixed effect model (\smallCap{lmm}) assumes a model
\begin{tightEquation*}
\by
	= \bX \bbeta
	+ \bZ \bm{\randEff} 
	+ \bm{\epsilon}, 
\end{tightEquation*}
where $\bbeta$ is a \textit{fixed} effect, $\thinnestspace \bm{\randEff} \sim \normalDist(\bm{0}, \RandEffVar)$ is a \textit{random} effect, and $\bm{\epsilon} \sim \normalDist(\bm{0}, \bm{R})$ is an error term.

\pause%
\medskip
Random effects are called as such b/c their values are assumed to be drawn from some distribution, rather than unknown-but-fixed values.
\end{frame}


\begin{frame}
\frametitle{\frametitleText}
% TODO: check terminology
The matrix $\bZ$ often represents group effects through a structure
\begin{tightEquation*}
\bZ = \begin{bmatrix}
	1 & 0 & 0 & \ldots & 0 \\
	1 & 0 & 0 & \ldots & 0 \\
	1 & 0 & 0 & \ldots & 0 \\
	0 & 1 & 0 & \ldots & 0 \\
	0 & 1 & 0 & \ldots & 0 \\
	0 & 0 & 1 & \ldots & 0 \\
	\vdots & & & \ddots &  
\end{bmatrix}.
\end{tightEquation*}

The variance $\RandEffVar$ is often block-diagonal w/ values $\thinnerspace \sigma_{\mathrm{rnd}, 1}^2, \thinnerspace \ldots, \sigma_{\mathrm{rnd}, J}^2$.
\end{frame}


\renewcommand{\frametitleText}{Hierarchical perspective}
\begin{frame}
\frametitle{\frametitleText}
In hierarchical modeling language, \smallCap{lmm} can be expressed as
\begin{tightEquation*}
\begin{aligned}
\by 
	\, \thinnerspace | \,  \thinnerspace 
	\bbeta, \bm{\randEff}, \bm{R}
	&\sim \normalDist\bigl(
		\bX \bbeta + \bZ \bm{\randEff}, 
		\bm{R} 
	\bigr), \\
\bm{\randEff}
	\, \thinnerspace | \,  \thinnerspace 
	\RandEffVar
	&\sim \normalDist\bigl(
		\bm{0}, \RandEffVar
	\bigr),
\end{aligned}
\end{tightEquation*}
where we've abbreviated explicit conditioning on observed $\bX$ and $\bZ$.

\pause
\medskip
Note that 
$\thinnerspace \bZ \bm{\randEff} 
	\, \thinnerspace | \, 
	\RandEffVar
	\sim \normalDist\bigl(
		\bm{0}, \bZ \RandEffVar \bZ^\transpose
	\bigr) \thinnerspace$
and hence
\begin{tightEquation*}
\bZ \bm{\randEff} + \epsilon
	\, \thinnerspace | \, 
	\RandEffVar, \bm{R}
	\sim \normalDist\bigl(
		\bm{0}, 
		\bZ \RandEffVar \bZ^\transpose + \bm{R}
	\bigr).
\end{tightEquation*}

\pause
So, after integrating out $\bm{\randEff}$, \smallCap{lmm} is given in its marginal form as
\begin{tightEquation*}
\by 
	\, \thinnerspace | \,  \thinnerspace 
	\bbeta, \RandEffVar, \bm{R}
	\sim \normalDist\bigl(
		\bX \bbeta, \thinnerspace
		\bm{V} \defeq \bZ \RandEffVar \bZ^\transpose + \bm{R}
	\bigr).
%	\ \text{ for } \
%	\bm{V} 
\end{tightEquation*}
or simply 
$\by 
	\, \thinnerspace | \,  \thinnerspace 
	\bbeta, \bm{V}
	\sim \normalDist\bigl(
		\bX \bbeta, \thinnerspace
		\bm{V}
	\bigr)$
in terms of $\bm{V} \defeq \bZ \RandEffVar \bZ^\transpose + \bm{R}$.
\end{frame}


\renewcommand{\frametitleText}{Frequentist estimation frameworks}
\begin{frame}
\frametitle{\frametitleText}
The (marginal) \smallCap{lmm} likelihood is thus given by, up to additive const, 
\begin{tightEquation*}
\log \likelihood (
	\by 
	\, \thinnerspace | \,  \thinnerspace 
	\bbeta, \bm{V}
) 
	= -  \frac12 \log | \bm{V} |
	- \frac12 (\by - \bX \bbeta)^\transpose \bm{V}^{-1} (\by - \bX \bbeta).
\end{tightEquation*}

\pause
\smallskip
If $\bm{V}$ were known, 
%we can obtain the \smallCap{MLE} of $\bbeta$ as 
we can estimate $\bbeta$ by maximizing the above via
\begin{tightEquation*}
\lse{\bbeta}
	= (\bX^\transpose \bm{V}^{-1} \bX)^{-1} \bX^\transpose \bm{V}^{-1} \by,
\end{tightEquation*}
and quantify uncertainty as, noting that $\variance\bigl( \by \, | \, \widehat{\bbeta}, \bm{V} \bigr) = \bm{V}$,
\begin{tightEquation*}
\variance( \lse{\bbeta} \given \bm{V} )
	= (\bX^\transpose \bm{V}^{-1} \bX)^{-1}.	
\end{tightEquation*}

\pause
Given this, we could estimate $\bm{V}$ by maximizing 
$\thinnerspace \log \likelihood (
	\by 
	\, \thinnerspace | \,  \thinnerspace 
	\lse{\bbeta}, \bm{V}
)$,
and then estimate $\bbeta$ by plugging in $\mle{\bm{V}}$ to the formula for $\lse{\bbeta}$.
\end{frame}


\begin{frame}
\frametitle{\frametitleText}
The above maximum likelihood approach turns out sub-optimal as it doesn't account for the degrees of freedom spent on estimating $\bbeta$. 

\pause
\smallskip
In particular, $\mle{\bm{V}}$ (and hence $\variance( \lse{\bbeta} \given \mle{\bm{V}} )$) underestimates the true variance as a consequence of ignoring uncertainty in $\lse{\bbeta}$.

\pause
\smallskip
This variance underestimation could also have downstream impacts, e.g. on the best linear unbiased prediction of the random effect $\bm{\randEff}$ via%
\begin{tightEquation*}
\blup{\bm{\randEff}}
	= \bm{D} \bZ^\transpose \bm{V}^{-1} (\by - \bX \lse{\bbeta}).
\end{tightEquation*}
\end{frame}


\begin{frame}
\frametitle{\frametitleText}
Alternative that avoids the variance underestimation is the \textit{restricted maximum likelihood} (\smallCap{reml}), maximizing the \textit{restricted likelihood}
\begin{tightEquation*}
\begin{aligned}
\log \likelihood_\remlSubscr(
	\by 
	\, \thinnerspace | \,  \thinnerspace 
	\bm{V}
) 
	= &- \frac12 \log | \bX^\transpose \bm{V}^{-1} \bX | \\
	&\hspace*{1em}-  \frac12 \log | \bm{V} |
	- \frac12 (\by - \bX \lse{\bbeta})^\transpose \bm{V}^{-1} (\by - \bX \lse{\bbeta}) \\
	= &- \frac12 \log | \bX^\transpose \bm{V}^{-1} \bX | 
	+ \log \likelihood (
		\by 
		\, \thinnerspace | \,  \thinnerspace 
		\lse{\bbeta}, \bm{V}
	).
\end{aligned}
\end{tightEquation*}

\pause
\medskip
As we show now, it turns out \smallCap{reml} admits a Bayesian interpretation, which provides an insight as to why it might be preferable to \smallCap{ml}.
\end{frame}


\begin{frame}
\frametitle{\frametitleText}
\textbf{Remark:}\\
Above expression for \smallCap{reml} is equivalent to a more classical one  in terms of \textit{error contrasts}, part of the data unexplained by fixed effects.

(More precisely, error contrasts are $\thinnerspace \sampleSize - \nPred \thinnerspace$ linearly independent components of
the residual $\thinnerspace \by - \bX \widehat{\bbeta}_\mathrm{ols}$ for $\thinnerspace \widehat{\bbeta}_\mathrm{ols} = (\bX^\transpose \bX)^{-1} \bX^\transpose \by$.)

\end{frame}


\renewcommand{\frametitleText}{Bayesian interpretation of \titleSmallCap{reml}: Intuition}
\begin{frame}
\frametitle{\frametitleText}
So, the \smallCap{mle} underetimates $\bm{V}$ due to ignoring uncertainty in $\lse{\bbeta}$.\\
How would a Bayesian deal with such issues? Integrate out $\bbeta$!

\pause
\smallskip
In other words, rather than considering 
$\likelihood(
	\by \given \widehat{\bbeta}, \bm{V}
)$
conditional on $\widehat{\bbeta}$, we should marginalize (i.e.\ average the likelihood) over unknown $\bbeta$:%
\begin{tightEquation*}
\likelihood(
	\by \given \bm{V}
)	
	= \int 
		\likelihood(\by \given \bbeta, \bm{V}) \density(\bbeta) 
	\diff \bbeta.
\end{tightEquation*}

\pause
The marginal likelihood requires a specification of the prior $\density(\bbeta)$; \\
to avoid a subjective choice, let's consider using a flat prior $\density(\bbeta) \propto 1$.

\smallskip
More precisely, we will show that
\begin{tightEquation*}
\likelihood_\remlSubscr(
	\by \given \bm{V}
)	
	= (2 \pi)^{\frac{\nPred}{2}} \! 
	\int 
		\likelihood(\by \given \bbeta, \bm{V})
	\diff \bbeta.
\end{tightEquation*}
\end{frame}


\newcommand{\densityWidth}{w}
\begin{frame}
\frametitle{\frametitleText}
\textbf{Note:} For the purpose of estimating $\bm{V}$, the use of the flat prior can be justified as a limit under $\thinnerspace \beta_j \iidSim \unifDist(-\densityWidth/2, \densityWidth/2)$ as $\densityWidth \to \infty$.

\pause
\smallskip
While the marginal likelihood itself doesn't make sense b/c
\begin{tightEquation*}
\likelihood(
	\by \given \bm{V}, \densityWidth
)	
	= \frac{1}{\densityWidth^\nPred}
	\int_{[-\densityWidth/2, \densityWidth/2]^\nPred} 
		\likelihood(\by \given \bbeta, \bm{V}) 
	\diff \bbeta
	\to 0,
\end{tightEquation*}
the ratio of the marginal likelihoods does still make sense:
\begin{tightEquation*}
\frac{
	\likelihood(
		\by \given \bm{V}
	)	
}{
	\likelihood(
		\by \given \bm{V}'
	)	
}
	= \lim_{\densityWidth \to \infty}
		\frac{
			\likelihood(
				\by \given \bm{V}, \densityWidth
			)	
		}{
			\likelihood(
				\by \given \bm{V}', \densityWidth
			)	
		}.
\end{tightEquation*}
\end{frame}


\renewcommand{\frametitleText}{Bayesian interpretation of \titleSmallCap{reml}: Derivation}
\begin{frame}
\frametitle{\frametitleText}
Showing 
$\thinnerspace
\likelihood_\remlSubscr(
	\by \given \bm{V}
)	
	= (2 \pi)^{\frac{\nPred}{2}} \! 
	\int 
		\likelihood(\by \given \bbeta, \bm{V})
	\diff \bbeta
\thinnerspace$
amounts to showing
\begin{tightEquation*}
\begin{aligned}
&\frac{1}{(2 \pi)^{\frac{\nPred}{2}} | \bX^\transpose \bm{V}^{-1} \bX |^{\frac12}}
%\frac{1}{| \bm{V} |^{1/2}}
\exp \!\left(
	- \frac12 (\by - \bX \lse{\bbeta})^\transpose \bm{V}^{-1} (\by - \bX \lse{\bbeta})
\right) \\
&\hspace{1.5em}= \int 
%		\frac{1}{| \bm{V} |^{1/2}}
		\exp \!\left(
			- \frac12 (\by - \bX \bbeta)^\transpose \bm{V}^{-1} (\by - \bX \bbeta)
		\right)
	\diff \bbeta.
\end{aligned}
\end{tightEquation*}

\pause
\medskip
To establish the above equality, let $\thinnerspace \| \bm{v} \|_{\bm{V}^{-1}}^2 \defeq \bm{v}^\transpose \bm{V}^{-1} \bm{v} \thinnerspace$ and\\ note that ``completing the square''/``Pythagorian theorem'' tells us 
\begin{equation*}
\setlength{\abovedisplayskip}{.45\baselineskip}
\setlength{\belowdisplayskip}{.5\baselineskip}
\big\| \by - \bX \bbeta \big\|_{\bm{V}^{-1}}^2
	= \big\| \by - \bX \lse{\bbeta} \big\|_{\bm{V}^{-1}}^2
	+ \big\| \bX \lse{\bbeta} - \bX \bbeta \big\|_{\bm{V}^{-1}}^2,
\end{equation*}
where $\thinnerspace \lse{\bbeta} = (\bX^\transpose \bm{V}^{-1} \bX)^{-1} \bX^\transpose \bm{V}^{-1} \by = \argmin_{\bbeta} \big\| \by - \bX \bbeta \big\|_{\bm{V}^{-1}}^2$.
\end{frame}


\begin{frame}
\frametitle{\frametitleText}
\vspace*{-.2\baselineskip}
This means that
\begin{tightEquation*}
\begin{aligned}
&\exp \!\left(
	- \frac12 \big\| \by - \bX \bbeta \big\|_{\bm{V}^{-1}}^2
\right) \\
	&= \exp \!\left(
		- \frac12 \big\| \by - \bX \lse{\bbeta} \big\|_{\bm{V}^{-1}}^2
	\right)
	\exp \!\left(
		- \frac12 \big\| \bX \lse{\bbeta} - \bX \bbeta \big\|_{\bm{V}^{-1}}^2
	\right).
\end{aligned}
\end{tightEquation*}

\pause
\vspace*{-.2\baselineskip}
Noting
$\thinnerspace \big\| \bX \lse{\bbeta} - \bX \bbeta \big\|_{\bm{V}^{-1}}^2  \negthinnerspace = (\lse{\bbeta} - \bbeta)^\transpose \bX^\transpose \bm{V}^{-1} \bX (\lse{\bbeta} - \bbeta) \thinnerspace$
and%
\begin{tightEquation*}
\setlength{\belowdisplayskip}{0\baselineskip}
\begin{aligned}
&\int 
	\exp \!\left(
		- \frac12 (\lse{\bbeta} - \bbeta)^\transpose \bX^\transpose \bm{V}^{-1} \bX (\lse{\bbeta} - \bbeta)
	\right)
\diff \bbeta \\
	&\hspace*{2em}= (2 \pi)^{\frac{\nPred}{2}} | \bX^\transpose \bm{V}^{-1} \bX |^{-\frac12},
\end{aligned}
\end{tightEquation*}
\pause%
we can conclude
\begin{tightEquation*}
\begin{aligned}
&\int 
	\exp \!\left(
		- \frac12 \big\| \by - \bX \bbeta \big\|_{\bm{V}^{-1}}^2
	\right) 
\diff \bbeta \\
	&\hspace*{2em}= \exp \!\left(
		- \frac12 \big\| \by - \bX \lse{\bbeta} \big\|_{\bm{V}^{-1}}^2
	\right)
	(2 \pi)^{\frac{\nPred}{2}} | \bX^\transpose \bm{V}^{-1} \bX |^{-\frac12}.
\end{aligned}
\end{tightEquation*}
\end{frame}


\renewcommand{\frametitleText}{Bayesian interpretation of \titleSmallCap{reml}: Takeaway}
\begin{frame}
\frametitle{\frametitleText}
R\smallCap{eml} can be interpreted as maximization of the marginal likelihood 
$\likelihood(
	\by \given \bm{V}
)	
	= 
	\int 
		\likelihood(\by \given \bbeta, \bm{V})
	\diff \bbeta \thinnerspace$
under the flat prior $\thinnerspace \density(\bbeta) \propto 1$.

%From this perspective, it is unsurprising that the resulting estimator of $\bm{V}$ avoids the underestimation caused when maximizing $\likelihood(\by \given \lse{\bbeta}, \bm{V})$.

\pause
\medskip
Yet another example of Bayesian thinking yielding a procedure with provable frequentist properties;
implications for (non-)Bayesians are:%
\pause%
\vspace{-.3\baselineskip}%
\begin{itemize}
\addtolength\itemsep{.08\baselineskip}
\item When faced with complex and/or limited data, consider full Bayes, w/ confidence, for practical (not philosophical) reasons.
\pause%
\item When Bayes makes less sense, whether for computational or political reasons, it might still help you devise a useful procedure.\\[3pt]
	\mbox{(e.g.\ Classical \smallCap{reml} derivations are rather involved and opaque.)}
\end{itemize}
\end{frame}


\renewcommand{\frametitleText}{Bayesian estimation framework}
\begin{frame}
\frametitle{\frametitleText}
Bayesian inference of \smallCap{lmm} is conceptually straightforward.

\pause%
\smallskip
The hierarchical model
\begin{tightEquation*}
\begin{aligned}
\by 
	\, \thinnerspace | \,  \thinnerspace 
	\bbeta, \bm{\randEff}, \bm{R}
	&\sim \normalDist\bigl(
		\bX \bbeta + \bZ \bm{\randEff}, 
		\bm{R} 
	\bigr), \\
\bm{\randEff}
	\, \thinnerspace | \,  \thinnerspace 
	\RandEffVar
	&\sim \normalDist\bigl(
		\bm{0}, \RandEffVar
	\bigr)
\end{aligned}
\end{tightEquation*}
dictates the likelihood 
$\thinnerspace 
\likelihood(\by, \bm{\randEff} \given \bbeta, \bm{\RandEffVar}, \bm{R}) 
	= \likelihood(\by \given \bbeta, \bm{\randEff}, \bm{R}) 
		\likelihood(\bm{\randEff} \given \bm{\RandEffVar}) $;
you only need to specify a prior on 
$\thinnerspace \bbeta, \bm{\RandEffVar}, \bm{R} \thinnerspace$ 
and apply Bayes' rule.

\pause%
\smallskip
And the posterior 
$\thinnerspace
\density(
	\bbeta, \bm{\randEff}, \bm{\RandEffVar}, \bm{R}
	\given 
	\by
) \propto 
	\likelihood(\by, \bm{\randEff} \given \bbeta, \bm{\RandEffVar}, \bm{R})
	\density(\bbeta, \bm{\RandEffVar}, \bm{R})$
\tightSpacing[.48ex]{gives all you need for point estimation, uncertainty quantification, etc.}

\pause%
\smallskip
Prior specification, especially for $\bm{\RandEffVar}$, warrants some thoughts;\\
we'll discuss key considerations and provide practical guidance.
\end{frame}


\section*{%
	\centerline{Bayesian inference of mixed effect models}%
}%

\begin{frame}
\frametitle{\frametitleText}
As pointed out previously, the only additional piece needed for Bayesian inference of \smallCap{lmm} is the prior specification on $\thinnerspace \bbeta, \bm{\RandEffVar}, \bm{R}$. 

And existing software packages like \texttt{brms} (Bayesian regression models using Stan) make the posterior computation straightforward. 

\medskip
Inference on $\bbeta$ generally isn't sensitive to a choice of prior, so weak/uninformative priors are fine, including the flat prior $\density(\bbeta) \propto 1$.

\tightSpacing[.48ex]{%
Residual cov $\bm{R}$ is often modeled as diagonal or $\sigma_\mathrm{res}^2 \Id$, the idea being that correlation in $\by$ has been captured by the random effect term $\bZ \bm{\randEff}$.
}%

\medskip
\tightSpacing[.48ex]{%
The remaining question is how to choose a prior for $\bm{\RandEffVar}$;
let's start by studying how it affects the posterior behavior in a simple case of \smallCap{lmm}.
}%
\end{frame}

\renewcommand{\frametitleText}{Prior on hierarchical variances: case study}
\begin{frame}
\frametitle{\frametitleText}
Recall the random group-mean model:
\begin{tightEquation*}
y_{ij} \given \mu_j
	\sim \normalDist(\mu_j, \sigma^2), \ \,
\mu_j \given \sigma_{\textrm{grp}} 
	\sim \normalDist( 
		\mu_\mathrm{grp}, \thinnerspace 
		\sigma_{\textrm{grp}}^2
	),
\end{tightEquation*}
for $\thinnerspace i = 1, \ldots, \sampleSize_j \thinnestspace$ and $\thinnerspace j = 1, \ldots, J$. 

\medskip
Consider two seemingly reasonable ``uninformative'' priors on $\sigma_{\textrm{grp}}$:
\begin{tightEnumerate}
\item $ \density( \sigma_{\textrm{grp}} ) \propto 1 \thinnerspace$;
\item $ \density( \sigma_{\textrm{grp}} ) \propto 1 / \sigma_{\textrm{grp}}$, which is the Jeffrey's/reference prior for scale families and is equivalent to $\thinnerspace \density( \log \sigma_{\textrm{grp}} ) \propto 1 \thinnerspace$ on $\thinnerspace (-\infty, \infty)$.
\end{tightEnumerate}

As it turns out, both choices are problematic.
\end{frame}

\begin{frame}
\frametitle{\frametitleText}
To analyze the posterior behavior, let's calculate the marginal likelihood of $\by$ given $\sigma_\mathrm{grp}$
%
--- note that, for $\thinnerspace \bar{y}_{j}  = n_j^{-1} \sum_{i = 1}^{n_j} y_{ij}$,
\begin{tightEquation*}
\setlength{\abovedisplayskip}{.4\baselineskip}%
\setlength{\belowdisplayskip}{.2\baselineskip}%
\prod_{i,j} 
	\normalDist\!\left( 
		y_{ij} \given \mu_j, \sigma^2
	\right)
\propto 
	\prod_{i,j} 
		\normalDist\!\left( 
			y_{ij} \given \bar{y}_{j}, \sigma^2
		\right)
	\prod_{j} 
		\normalDist\!\left( 
			\bar{y}_{j} \given \mu_j, \sigma^2 / n_j
		\right)
\end{tightEquation*}
because
$\thinnerspace
\sum_{i,j} ( y_{ij} - \mu_j )^2
	= \sum_{i,j} ( y_{ij} - \bar{y}_{j} )^2
	+ \sum_{j} n_j ( \bar{y}_{j} - \mu_j )^2
$.

\bigskip
For the purpose of estimating $\sigma_\mathrm{grp}$, the model is thus equivalent to
\begin{tightEquation*}
\bar{y}_{j} \given \mu_j
	\sim \normalDist(\mu_j, \sigma^2 / n_j), \ \,
\mu_j \given \sigma_{\textrm{grp}} 
	\sim \normalDist( 
		\mu_\mathrm{grp}, \thinnerspace 
		\sigma_{\textrm{grp}}^2
	).
\end{tightEquation*}
Marginalizing over $\mu_j$'s, therefore, we obtain 
\begin{tightEquation*}
\bar{y}_{j} \given \sigma_{\textrm{grp}} 
	\sim \normalDist( 
		\mu_\mathrm{grp}, \thinnerspace 
		\sigma^2 / n_j + \sigma_{\textrm{grp}}^2
	).
\end{tightEquation*}
\end{frame}

\begin{frame}
\frametitle{\frametitleText}
Under the flat prior $\thinnerspace \density(\sigma_\mathrm{grp}) \propto 1$, the posterior is
\begin{tightEquation*}
\density( \sigma_\mathrm{grp} \given \by )
	\propto 
	\prod_{j = 1}^J \frac{1}{( \sigma^2 / n_j + \sigma_\mathrm{grp}^2 )^{1/2}}
		\exp \left(
			- \frac12 \frac{
				(\bar{y}_{j} - \mu_j)^2
			}{
				\sigma^2 / n_j + \sigma_\mathrm{grp}^2
			}
		\right).
\end{tightEquation*}

As $\sigma_\mathrm{grp} \to \infty$, we have
%For $\thinnerspace \sigma_\mathrm{grp} \gg \sigma^2 / n_j$, we have
\begin{tightEquation*}
\density( \sigma_\mathrm{grp} \given \by )
	\sim \prod_{j = 1}^J \frac{1}{( \sigma_\mathrm{grp}^2 )^{1/2}}
	\sim \frac{1}{\sigma_\mathrm{grp}^{J}}.
\end{tightEquation*}

The posterior is improper ($\int_0^\infty \density( \sigma_\mathrm{grp} \given \by ) \diff \sigma_\mathrm{grp} = \infty$) unless $J \geq 3$.

\medskip
\tightSpacing[.5ex]{%
This makes an intuitive sense as we need $J \geq 2$ to estimate variance among $\mu_j$'s. 
And, as $\mu_j$'s are only indirectly observed, we need $J > 2$.
}%
\end{frame}

\begin{frame}
\frametitle{\frametitleText}
Another feature of the marginal likelihood 
\begin{tightEquation*}
\bar{y}_{j} \given \sigma_{\textrm{grp}} 
	\sim \normalDist( 
		\mu_\mathrm{grp}, \thinnerspace 
		\sigma^2 / n_j + \sigma_{\textrm{grp}}^2
	)
\end{tightEquation*}
is
$\thinnerspace \likelihood( \by \given \sigma_\mathrm{grp} = 0) > 0$;
i.e.\ can't eliminate the possibility of $\sigma_\mathrm{grp} = 0$.

\medskip
Under the prior 
$\density( \sigma_{\textrm{grp}} ) \propto 1 / \sigma_{\textrm{grp}} $,
the posterior is always improper!
\end{frame}

\begin{frame}
\frametitle{\frametitleText}
Shortcoming of the priors $\thinnerspace \density( \sigma_{\textrm{grp}} ) \propto 1 \thinnerspace$ and $\thinnerspace \density( \sigma_{\textrm{grp}} ) \propto 1 / \sigma_{\textrm{grp}}$ reflects that $\sigma_{\textrm{grp}}$ is only weakly identifiable from data, esp.\ when $J$ is small.

\tightSpacing[.525ex]{%
Also reflects how ``uninformative'' priors are unexpectedly informative --- they tend to place a lot of prior prob on unrealistic param values.
}%

\medskip
To see the issue, note how $\density( \sigma_\mathrm{grp} ) \propto 1$ places ``infinitely more prior belief'' on $\sigma_\mathrm{grp} > r$ than  $\sigma_\mathrm{grp} \leq r$ regardless of how large $r$ is b/c
\begin{tightEquation*}
\int_0^r \density( \sigma_\mathrm{grp} ) \diff \sigma_\mathrm{grp} < \infty
	\ \ \text{ while } \
	\int_r^\infty \density( \sigma_\mathrm{grp} ) \diff \sigma_\mathrm{grp} = \infty.
\end{tightEquation*}

Similarly, $\density( \sigma_\mathrm{grp} ) \propto 1 / \sigma_\mathrm{grp}$ places infinitely more prior belief on 
%$\sigma_\mathrm{grp} < r^{-1}$ and $\sigma_\mathrm{grp} > r$ than on $r^{-1} \leq \sigma_\mathrm{grp} \leq r$.
$\sigma_\mathrm{grp} \in [0, r^{-1}) \cup (r, \infty) \thinnerspace$ than on $\thinnerspace \sigma_\mathrm{grp} \in [r^{-1}, r]$.
\end{frame}

\begin{frame}
\frametitle{\frametitleText}

\newcommand{\corrMat}{\bm{\Gamma}}
\renewcommand{\frametitleText}{Prior on variance of random effects}
\begin{frame}
\frametitle{\frametitleText}
More generally, for the \smallCap{lmm}
\begin{tightEquation*}
\begin{aligned}
\by = \bX \bbeta + \bZ \bm{\randEff} + \bm{\epsilon}
\ \text{ with } \,
\bm{\randEff}
	\sim \normalDist\bigl(
		\bm{0}, \RandEffVar
	\bigr),
\end{aligned}
\end{tightEquation*}
how to choose a prior on $\bm{\RandEffVar}$ (beyond the case $\bZ = \bindicator$, $\bm{\RandEffVar} = \sigma_\mathrm{grp}^2 \Id$)?

\smallskip
In the absence of structural info, a reasonable prior on $\bm{\RandEffVar}$ can be built from the \textit{\smallCap{LKJ} distribution} for correlation matrices.\\
\hfill (\smallCap{LKJ} = Lewandowski-Kurowicka-Joe)

\smallskip
More precisely, we parametrize as 
$ \randEffVar_{ij} = \sigma_i \sigma_j \rho_{ij} $
and place on $[\rho_{ij}]_{i, j}$ the \smallCap{LKJ} prior w/ shape param $\eta$ ($= 1$ providing \texttt{brms} default).

Priors on $\sigma_i$ can be chosen e.g.\ as $\operatorname{Student}^+(\mathrm{df} = 3)$ (\texttt{brms} default).\\
\hfill \citep{burkner2017brms}
\end{frame}


\begin{frame}
\frametitle{\frametitleText}
L\smallCap{KJ} prior is reasonable in a sense that it 
\begin{tightItemize}
\item has the identity matrix as its mean;
\item is uniform on the space of correlation matrices when $\eta = 1$;
\item has marginal $\randEffVar_{ij} \sim 2 \thinspace \betaDist \!\left( \eta - 1 + \frac{\nRandEff}{2},  \eta - 1 + \frac{\nRandEff}{2} \right) - 1$, mildly shrinking towards zero correlations as the matrix size $\nRandEff$ grows.
\end{tightItemize}
\end{frame}
